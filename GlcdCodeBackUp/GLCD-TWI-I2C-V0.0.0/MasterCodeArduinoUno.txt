#include <Wire.h>

#define ATMEGA8_ADDR 0x08

#define CMD_STRING8   0x01 // max 18 caracters
#define CMD_STRING16  0x02 // max 14 characters
#define CMD_CLEAR     0x03
#define CMD_16by16icon  0x04

void setup()
{
    Serial.begin(9600);

    Wire.begin();   // Arduino = I2C Master

    delay(1000);

    Serial.println("I2C Master Started");
    sendGLCDString(CMD_CLEAR, 0xff, 0xff, ""); // dummy cmd if the i2c in the slave was stuck
    sendGLCDString(CMD_CLEAR, 0xff, 0xff, "");
}

void sendGLCDString(uint8_t command,
                    uint8_t row,
                    uint8_t column,
                    const char *message)
{
    Wire.beginTransmission(ATMEGA8_ADDR);

    // Byte 0: Command
    Wire.write(command);

    // Byte 1: Row
    Wire.write(row);

    // Byte 2: Column
    Wire.write(column);

    // Byte 3 onwards: Message
    Wire.write((const uint8_t *)message, strlen(message));

    byte error = Wire.endTransmission(); 
    delay(20); //Minimum possible delay 20 for the glcd to work properly
}

}
